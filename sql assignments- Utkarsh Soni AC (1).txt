-- 1. List all the columns of the Salespeople table.
SELECT * FROM Salespeople;




-- 2. List all customers with a rating of 100.
SELECT *
FROM Customers
WHERE RATING = 100;




-- 3. Find customers with NULL values in the city column.
SELECT *
FROM Customers

-- 4. Find the largest order taken by each salesperson on each date.
SELECT SNUM, ODATE, MAX(AMT) AS MAX_AMOUNT
FROM Orders
GROUP BY SNUM, ODATE;




-- 5. Arrange Orders by descending customer number.

FROM Orders

-- 6. Find salespeople who currently have orders.
SELECT DISTINCT s.*
FROM Salespeople s
JOIN Orders o ON s.SNUM = o.SNUM;




-- 7. List customer names with the salespeople serving them.

FROM Customers c

-- 8. Names and numbers of salespeople having more than one customer.
SELECT s.SNAME, s.SNUM
FROM Salespeople s
JOIN Customers c ON s.SNUM = c.SNUM
GROUP BY s.SNUM, s.SNAME
HAVING COUNT(c.CNUM) > 1;




-- 9. Count orders of each salesperson in descending order.

-- 100. Does the customer with the most orders have the maximum rating?
-- Returns the customer(s) with the highest order count and their rating.

FROM Customers c
JOIN Orders o ON c.CNUM = o.CNUM
GROUP BY c.CNUM, c.CNAME, c.RATING
HAVING COUNT(o.ONUM) = (
    SELECT MAX(order_count)
    FROM (
        SELECT COUNT(*) AS order_count
        FROM Orders
        GROUP BY CNUM
    ) x
);
